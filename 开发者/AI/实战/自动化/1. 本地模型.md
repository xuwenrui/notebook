#### 安装ollama
Ollama自定义安装路径
OllamaSetup.exe /DIR="D:\tools\Ollama"

#### ollama下载模型（建议使用）
setx OLLAMA_MODELS "D:\tools\Ollama\models"
ollama run SimonPu/Qwen3-Coder:30B-Instruct_Q4_K_XL
#### modelscop下载模型
1. conda activate ai-develop
2. pip install modelscop
3. # 下载到 D 盘指定目录（直接保存，不经过缓存）
modelscope download --model Qwen/Qwen3-Coder-Next-GGUF --local_dir D:\tools\Ollama\models\qwen3-coder-next-gguf



选择 **Qwen2.5-Coder-7B-Q4_K_M**（4.5GB）
modelscope download --model qwen/Qwen2.5-Coder-7B-Instruct-GGUF --local_dir D:\tools\Ollama\models\qwen25-7b
[Qwen2.5-Coder-14B-Instruct-GGUF](https://modelscope.cn/models/qwen/Qwen2.5-Coder-14B-Instruct-GGUF)
modelscope download --model qwen/Qwen2.5-Coder-14B-Instruct-GGUF --local_dir D:\tools\Ollama\models\qwen25-14b



ollama加载模型
```
# Modelfile 内容（严格按此配置）
FROM D:/tools/Ollama/models/qwen25-14b/qwen2.5-coder-14b-instruct-q4_k_m.gguf

# 专为代码模型优化的系统提示词
SYSTEM """
You are Qwen2.5-Coder, an expert AI programmer developed by Alibaba Cloud.
Your primary role is to write, explain, and debug code in ALL programming languages.
NEVER provide non-technical answers. If asked non-coding questions, respond: "I am a code specialist. Please ask about programming."
"""

# 关键参数（针对代码模型优化）
PARAMETER num_ctx 8192    # 更长上下文（适合代码）
PARAMETER temperature 0.2  # 降低随机性（代码需精确）
PARAMETER repeat_penalty 1.1
```


ollama create qwen25-coder-14b -f .\Modelfile
检查模型列表
ollama list
测试代码生成能力（正确用法）
ollama run qwen25-coder-14b "用 Python 写一个快速排序函数"
